#### 步骤 1.2：实现 ModelRegistry

**新文件**: `app/src/ai/models/ModelRegistry.ts`

```typescript
export class ModelRegistry {
  private static instance: ModelRegistry;
  
  // 三层模型数据
  private builtinModels: Map<string, ModelConfig> = new Map();
  private userModels: Map<string, ModelConfig> = new Map();
  private discoveredModels: Map<string, ModelConfig> = new Map();
  
  // Provider 配置
  private providerConfigs: Map<string, ProviderConfig> = new Map();
  
  private constructor() {
    this.loadBuiltinModels();
  }
  
  static getInstance(): ModelRegistry {
    if (!ModelRegistry.instance) {
      ModelRegistry.instance = new ModelRegistry();
    }
    return ModelRegistry.instance;
  }
  
  /** 加载内置模型（从 ModelConfigs.ts） */
  private loadBuiltinModels(): void {
    for (const [key, config] of Object.entries(ALL_MODEL_CONFIGS)) {
      if (config.firstParty) {
        this.builtinModels.set(config.firstParty, config);
      }
    }
  }
  
  /** 从配置文件中加载用户配置（覆盖/新增） */
  loadUserConfigs(): void {
    const providersCfg = loadProvidersConfig();
    const modelsCfg = loadModelsConfig();
    
    // 加载 Provider 配置
    for (const [id, cfg] of Object.entries(providersCfg.providers)) {
      this.providerConfigs.set(id, cfg);
    }
    
    // 加载模型覆盖/自定义
    for (const [modelId, override] of Object.entries(modelsCfg.models)) {
      if (override.baseModel && this.builtinModels.has(override.baseModel)) {
        // 基于 baseModel 的自定义模型
        const base = this.builtinModels.get(override.baseModel)!;
        this.userModels.set(modelId, {
          ...base,
          firstParty: modelId,
          displayName: override.displayName ?? base.displayName,
          ...(override.contextWindow && { contextWindow: override.contextWindow }),
          ...(override.capabilities && { capabilities: override.capabilities as ModelCapability[] }),
        });
      } else {
        // 覆盖现有模型或新增
        const existing = this.builtinModels.get(modelId);
        this.userModels.set(modelId, {
          ...existing ?? {} as ModelConfig,
          ...override,
          firstParty: modelId,
        });
      }
    }
  }
  
  /** 获取合并后的完整模型列表 */
  getAllModels(): ModelConfig[] {
    const result = new Map(this.builtinModels);
    // 用户配置覆盖内置
    for (const [id, cfg] of this.userModels) {
      result.set(id, cfg);
    }
    // 运行时发现追加
    for (const [id, cfg] of this.discoveredModels) {
      if (!result.has(id)) {
        result.set(id, cfg);
      }
    }
    return Array.from(result.values());
  }
  
  /** 获取指定模型配置 */
  getModel(modelId: string): ModelConfig | undefined {
    return this.userModels.get(modelId)
      ?? this.discoveredModels.get(modelId)
      ?? this.builtinModels.get(modelId);
  }
  
  /** 获取 Provider 配置 */
  getProviderConfig(providerId: string): ProviderConfig | undefined {
    return this.providerConfigs.get(providerId);
  }
  
  /** 运行时发现模型（Ollama 等） */
  discoverModel(modelId: string, config: Partial<ModelConfig>): void {
    this.discoveredModels.set(modelId, {
      firstParty: modelId,
      displayName: modelId,
      contextWindow: 200000,
      maxOutputTokens: 4096,
      ...config,
    } as ModelConfig);
  }
}
```

## 定价同步机制（贯穿 Sprint 1-2）

> 实现用户可维护、社区可同步、内置默认值的三层定价体系。

### 为什么不能直接实时查价

经过调研，截至 2025 年各主要供应商**均不提供标准化定价 API**：

| 供应商 | 定价获取方式 | 是否有程序化 API |
|--------|-------------|----------------|
| OpenAI | 网页 `openai.com/api/pricing` | ❌ 无定价 API，`/v1/models` 仅返回模型 ID |
| Anthropic | 网页 `claude.com/pricing` | ❌ 仅有用量查询 API（查自己的历史消费） |
| DeepSeek | 网页 `api-docs.deepseek.com` | ❌ 无定价 API |
| Google Gemini | 网页 `cloud.google.com` | ❌ 无定价 API |
| Azure OpenAI | 网页 + 价格计算器 | ✅ 有结构化定价页面（非 REST API） |

定价市场 2025 的特征：
- 价格在持续下降（年内降幅 83-90%），变动频繁
- 各厂商定价发布在网页上，无统一 API
- 同一模型按上下文长度、缓存命中、批量/实时等条件价格不同

### 三层定价机制

```
┌─────────────────────────────────────────────────┐
│              三层定价机制                         │
├─────────────────────────────────────────────────┤
│                                                   │
│  第一层：内置默认定价（编译时）                     │
│    → ModelConfigs.ts 中维护，随发版更新            │
│    → 用户无配置时自动使用                          │
│                                                   │
│  第二层：用户配置文件覆盖（运行时）                 │
│    → ~/.pyapp/pricing.yaml                        │
│    → 用户可手工修改或通过管理 UI 编辑              │
│    → 优先级高于内置默认值                          │
│                                                   │
│  第三层：社区定价源同步（可选，定时拉取）           │
│    → 从 GitHub 社区维护的定价 JSON 拉取            │
│    → 命令 /pricing sync 手动触发                   │
│    → 优先级最高，但可通过配置文件降级              │
│                                                   │
└─────────────────────────────────────────────────┘
```

#### 扩展 ModelRegistry

```typescript
// ModelRegistry 新增定价相关方法

class ModelRegistry {
  // 定价的三层来源
  private builtinPricing: Map<string, ModelConfig['pricing']>;
  private userPricing: Map<string, PricingOverride>;     // 从 pricing.yaml 加载
  private syncedPricing: Map<string, PricingOverride>;   // 从社区同步
  private pricingCachePath: string;                       // ~/.pyapp/cache/pricing.json

  /**
   * 获取模型的最终定价（三层合并）
   * 优先级：用户覆盖 > 社区同步 > 内置默认
   */
  getModelPricing(modelId: string): { inputPer1M: number; outputPer1M: number } | null {
    // 1. 用户覆盖
    const user = this.userPricing.get(modelId);
    if (user) return user;

    // 2. 社区同步
    const synced = this.syncedPricing.get(modelId);
    if (synced) return synced;

    // 3. 内置默认（从模型配置读取）
    const model = this.getModel(modelId);
    if (model?.pricing) return model.pricing;

    return null;
  }

  /**
   * 从社区源同步定价
   */
  async syncPricing(sourceUrl?: string): Promise<number> {
    const url = sourceUrl ?? DEFAULT_PRICING_SOURCE;
    const response = await fetch(url);
    const data = await response.json();

    let count = 0;
    for (const [modelId, pricing] of Object.entries(data.pricing)) {
      this.syncedPricing.set(modelId, pricing as PricingOverride);
      count++;
    }

    // 缓存到本地
    const cacheDir = join(resolvePyappHome(), 'cache');
    if (!existsSync(cacheDir)) mkdirSync(cacheDir, { recursive: true });
    writeFileSync(this.pricingCachePath, JSON.stringify(data), 'utf-8');

    return count;
  }

  /**
   * 从缓存加载上次同步的定价
   */
  private loadSyncedPricingCache(): void {
    if (existsSync(this.pricingCachePath)) {
      const data = JSON.parse(readFileSync(this.pricingCachePath, 'utf-8'));
      for (const [modelId, pricing] of Object.entries(data.pricing)) {
        this.syncedPricing.set(modelId, pricing as PricingOverride);
      }
    }
  }

  /**
   * 从用户配置文件加载定价覆盖
   */
  private loadUserPricing(): void {
    const config = loadPricingConfig();
    for (const [modelId, pricing] of Object.entries(config.pricing)) {
      this.userPricing.set(modelId, pricing);
    }
  }
}
```

## 风险与缓解

| 风险 | 等级 | 缓解措施 |
|------|------|---------|
| YAML 依赖引入后 Bun 编译兼容性 | 中 | Bun 原生支持 js-yaml，先在 `bun test` 中验证 |
| 20 个文件引用 ModelConfigs，回归面广 | 高 | 每次修改后运行 `bun run typecheck` + `bun test` |
| 配置加载时序（启动时 vs 运行时） | 中 | 启动时加载一次，提供 `reload` 方法支持运行时热重载 |
| API Key 明文存储安全 | 高 | 配置文件权限设置为 `600`，UI 显示 `********` |
| 用户迁移（现有硬编码→配置化） | 中 | 无 providers.yaml 时完全兼容旧逻辑，零迁移成本 |
